cmake_minimum_required(VERSION 3.25)
project(batchquay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BQ_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "This validation profile requires GCC or Clang")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
if(BQ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(bq src/model.cpp src/cache.cpp src/scheduler.cpp)
target_include_directories(bq PUBLIC src)
add_executable(batchquay src/main.cpp)
target_link_libraries(batchquay PRIVATE bq)
enable_testing()
add_executable(native_tests tests/native_tests.cpp)
target_link_libraries(native_tests PRIVATE bq)
add_test(NAME native COMMAND native_tests)
add_executable(bq_benchmark tools/benchmark_driver.cpp)
target_link_libraries(bq_benchmark PRIVATE bq)
if(WIN32)
  target_link_libraries(bq_benchmark PRIVATE psapi)
endif()
